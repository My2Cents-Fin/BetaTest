DO $$
DECLARE
  v_user_id uuid;
  v_household_id uuid;
BEGIN
  -- Find user by email
  SELECT id INTO v_user_id FROM auth.users WHERE email = '919638527410@my2cents.app';
  IF v_user_id IS NULL THEN
    RAISE EXCEPTION 'User not found for phone 9638527410';
  END IF;

  -- Find household
  SELECT household_id INTO v_household_id FROM household_members WHERE user_id = v_user_id;

  IF v_household_id IS NOT NULL THEN
    DELETE FROM household_sub_categories WHERE household_id = v_household_id;
    DELETE FROM household_members WHERE user_id = v_user_id;
    DELETE FROM households WHERE id = v_household_id;
  END IF;

  DELETE FROM user_pins WHERE user_id = v_user_id;
  DELETE FROM users WHERE id = v_user_id;
  DELETE FROM auth.users WHERE id = v_user_id;

  RAISE NOTICE 'Deleted user % and household %', v_user_id, v_household_id;
END $$;








SELECT 'auth.users' AS tbl, count(*) FROM auth.users WHERE email = '919638527410@my2cents.app'
UNION ALL
SELECT 'users', count(*) FROM users WHERE id IN (SELECT id FROM auth.users WHERE email = '919638527410@my2cents.app')
UNION ALL
SELECT 'user_pins', count(*) FROM user_pins WHERE user_id IN (SELECT id FROM auth.users WHERE email = '919638527410@my2cents.app')
UNION ALL
SELECT 'household_members', count(*) FROM household_members WHERE user_id IN (SELECT id FROM auth.users WHERE email = '919638527410@my2cents.app')
UNION ALL
SELECT 'households', count(*) FROM households WHERE id IN (SELECT household_id FROM household_members WHERE user_id IN (SELECT id FROM auth.users WHERE email = '919638527410@my2cents.app'));
